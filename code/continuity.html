continuity.py

Find continuity breaks in a sampled signal.

"""Find continuity breaks in a sampled signal.

A signal logged from a controller or monitor can hide jumps (C0 breaks) and
kinks (C1 breaks) from hard switches, clipping, thresholds and angle wrapping.
`find_breaks` locates them so they can be replaced with the modifiers in
`smooth.py`.

A break is anything the sampling cannot resolve as smooth: a rounded corner
whose radius is many samples wide is not reported, a sharper one is. Assumes
near-uniform sampling. For noisy data raise `abs_tol` to a few noise sigmas.
"""
from dataclasses import dataclass

import numpy as np


@dataclass(frozen=True)
class Break:
    index: int     # first sample after the jump (order 0) or the kink sample (order 1)
    time: float
    order: int     # 0 = jump in value, 1 = kink (jump in slope)
    size: float    # signed jump height, or signed change of slope


def _outliers(a, window, ratio, tol):
    """Indices where |a_k| exceeds ratio * (median |a| of its neighbours) + tol."""
    mag = np.abs(a)
    hits = []
    for k in range(len(mag)):
        lo, hi = max(0, k - window), min(len(mag), k + window + 1)
        neighbours = np.delete(mag[lo:hi], k - lo)
        ref = np.median(neighbours) if neighbours.size else 0.0
        if mag[k] > ratio * ref + tol:
            hits.append(k)
    return hits


def find_breaks(t, y, window=7, ratio=8.0, abs_tol=1e-9):
    t, y = np.asarray(t, dtype=float), np.asarray(y, dtype=float)
    dt = float(np.median(np.diff(t)))
    dy = np.diff(y)
    jumps = _outliers(dy, window, ratio, abs_tol)
    found = [Break(j + 1, float(t[j + 1]), 0, float(dy[j])) for j in jumps]

    slope = dy / dt
    dslope = np.diff(slope)           # dslope[k] is the slope change at sample k + 1
    masked = {j + off for j in jumps for off in (-1, 0)}   # a jump spikes two entries
    for k in _outliers(dslope, window, ratio, abs_tol / dt):
        if k not in masked:
            found.append(Break(k + 1, float(t[k + 1]), 1, float(dslope[k])))
    return _merge_adjacent(sorted(found, key=lambda b: (b.order, b.index)))


def _merge_adjacent(found):
    """A kink between two samples spreads over two neighbouring entries; join them
    into one break whose size is the total slope change."""
    merged = []
    for b in found:
        prev = merged[-1] if merged else None
        if prev and prev.order == b.order and b.index - prev.index <= 1:
            keep = prev if abs(prev.size) >= abs(b.size) else b
            merged[-1] = Break(keep.index, keep.time, b.order, prev.size + b.size)
        else:
            merged.append(b)
    return sorted(merged, key=lambda b: b.index)
