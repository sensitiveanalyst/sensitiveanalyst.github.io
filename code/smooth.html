smooth.py

Smooth replacements for kinks and discontinuities.

"""Smooth replacements for kinks and discontinuities.

Every function here is C-infinity and carries a width parameter (eps, width,
or r). Smaller values track the original non-smooth function more closely
and make derivatives larger (stiffer), so treat the width as a design choice
and, for shooting or MPC, lower it by continuation.

Replaces                      Use
|x|                           smooth_abs
sign(x), v/|v|                smooth_sign, smooth_normalize
max(x, 0)                     softplus
max_i / min_i                 smooth_max, smooth_min
clip(x, -a, a)                tanh_sat
if s > c: b else a            smooth_switch (C-infinity blend, exact outside a band)
bang-bang from |u| <= u_max   barrier_cost + barrier_control (a smooth *cost*, so the
                              minimum principle still applies to the smoothed problem)
STL always / eventually       smooth_always, smooth_eventually
"""
import numpy as np
from scipy.special import logsumexp, xlogy


def smooth_abs(x, eps):
    """sqrt(x^2 + eps^2) - eps. Zero at 0, and 0 <= |x| - value <= eps."""
    x = np.asarray(x, dtype=float)
    return np.sqrt(x * x + eps * eps) - eps


def smooth_sign(x, eps):
    """Derivative of sqrt(x^2 + eps^2); tends to sign(x) as eps -> 0."""
    x = np.asarray(x, dtype=float)
    return x / np.sqrt(x * x + eps * eps)


def smooth_normalize(v, eps):
    """v / sqrt(|v|^2 + eps^2). Defined at v = 0, unlike v / |v|."""
    v = np.asarray(v, dtype=float)
    return v / np.sqrt(v @ v + eps * eps)


def softplus(x, eps):
    """eps * log(1 + exp(x / eps)). max(x, 0) <= value <= max(x, 0) + eps ln 2."""
    return eps * np.logaddexp(0.0, np.asarray(x, dtype=float) / eps)


def smooth_max(values, eps, axis=-1):
    """Log-sum-exp. max <= value <= max + eps ln n, so it never under-estimates."""
    return eps * logsumexp(np.asarray(values, dtype=float) / eps, axis=axis)


def smooth_min(values, eps, axis=-1):
    """-smooth_max(-values). min - eps ln n <= value <= min, so it never over-estimates."""
    return -smooth_max(-np.asarray(values, dtype=float), eps, axis=axis)


def tanh_sat(x, limit, slope=1.0):
    """limit * tanh(slope * x / limit): bounded by limit, gain `slope` near zero.
    Smooth stand-in for clip(slope * x, -limit, limit)."""
    return limit * np.tanh(slope * np.asarray(x, dtype=float) / limit)


def bump_step(s):
    """C-infinity step: 0 for s <= 0, 1 for s >= 1, strictly monotone between.
    All derivatives vanish at both ends, so a blend built on it is *exactly*
    the old behaviour outside the transition band."""
    s = np.clip(np.asarray(s, dtype=float), 0.0, 1.0)
    pos = np.where(s > 0.0, s, 1.0)
    neg = np.where(s < 1.0, 1.0 - s, 1.0)
    a = np.where(s > 0.0, np.exp(-1.0 / pos), 0.0)
    b = np.where(s < 1.0, np.exp(-1.0 / neg), 0.0)
    return a / (a + b)


def smooth_switch(s, low, high, a, b):
    """Blend a -> b as the monitored quantity s crosses [low, high].
    Exactly a for s <= low, exactly b for s >= high. For control-affine systems
    and a shared Lyapunov function V, V-dot is affine in u, so if both a and b
    satisfy V-dot <= -gamma V then so does every blend."""
    w = bump_step((s - low) / (high - low))
    return (1.0 - w) * np.asarray(a, dtype=float) + w * np.asarray(b, dtype=float)


def barrier_cost(u, u_max, r=1.0):
    """Smooth strictly convex input cost for the box |u| <= u_max (elementwise).

        w(u) = (r u_max^2 / 2) [(1+s) ln(1+s) + (1-s) ln(1-s)],   s = u / u_max

    w(u) ~ r u^2 / 2 near zero, w'(u) = r u_max atanh(s), and w is finite at
    |s| = 1 with infinite slope there. Minimising w(u) + g u over u gives
    u = -u_max tanh(g / (r u_max)) (see barrier_control), so the Hamiltonian
    minimiser is smooth in g and never leaves the box. As r -> 0 it tends to
    the bang-bang law -u_max sign(g)."""
    s = np.clip(np.asarray(u, dtype=float) / u_max, -1.0, 1.0)
    return 0.5 * r * u_max**2 * (xlogy(1.0 + s, 1.0 + s) + xlogy(1.0 - s, 1.0 - s))


def barrier_control(g, u_max, r=1.0):
    """argmin_u  w(u) + g u  =  -u_max tanh(g / (r u_max))."""
    return -u_max * np.tanh(np.asarray(g, dtype=float) / (r * u_max))


def smooth_always(margins, eps):
    """Smooth robustness of 'always (h >= 0)' over samples: a soft minimum.
    Sound: smooth_always <= min, so smooth_always >= 0 implies every margin >= 0."""
    return smooth_min(margins, eps)


def smooth_eventually(margins, eps):
    """Smooth robustness of 'eventually (h >= 0)': a soft maximum (>= max)."""
    return smooth_max(margins, eps)
